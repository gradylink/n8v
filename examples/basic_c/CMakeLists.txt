add_executable(n8v-basic-c-example main.c)
if(TARGET n8v)
  target_link_libraries(n8v-basic-c-example PRIVATE n8v)
else()
  target_link_libraries(n8v-basic-c-example PRIVATE n8v_shared)
endif()
target_compile_definitions(n8v-basic-c-example PRIVATE N8V_EXAMPLE_ASSET_DIR="assets")
set_target_properties(n8v-basic-c-example PROPERTIES LINKER_LANGUAGE CXX)
if(EMSCRIPTEN)
  set_target_properties(n8v-basic-c-example PROPERTIES SUFFIX ".html")
endif()

n8v_embed_manifest(n8v-basic-c-example)
n8v_copy_runtime(n8v-basic-c-example)
