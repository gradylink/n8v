add_executable(n8v-basic-example main.cpp)
if(TARGET n8v)
  target_link_libraries(n8v-basic-example PRIVATE n8v)
else()
  target_link_libraries(n8v-basic-example PRIVATE n8v_shared)
endif()
if(EMSCRIPTEN)
  target_compile_definitions(n8v-basic-example PRIVATE N8V_EXAMPLE_ASSET_DIR="/assets")
  target_link_options(n8v-basic-example PRIVATE "SHELL:--preload-file ${CMAKE_CURRENT_SOURCE_DIR}/assets@/assets")
  set_target_properties(n8v-basic-example PROPERTIES SUFFIX ".html")
else()
  target_compile_definitions(n8v-basic-example PRIVATE N8V_EXAMPLE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()
